# --- configuration: flag > env > config file > default ----------------------
== raw-json env true
args: dns info
response: data:{"a":1}
env: SUPERVISOR_RAW_JSON=true
== raw-json env 1
args: dns info
response: data:{"a":1}
env: SUPERVISOR_RAW_JSON=1
== raw-json env garbage
args: dns info
response: data:{"a":1}
env: SUPERVISOR_RAW_JSON=yes
== raw-json env true, flag false
args: dns info --raw-json=false
response: data:{"a":1}
env: SUPERVISOR_RAW_JSON=true
== raw-json env empty
args: dns info
response: data:{"a":1}
env: SUPERVISOR_RAW_JSON=
== token: SUPERVISOR_API_TOKEN beats SUPERVISOR_TOKEN
args: dns info
env: SUPERVISOR_API_TOKEN=api-token-env
== token: flag beats env
args: dns info --api-token from-flag
env: SUPERVISOR_API_TOKEN=api-token-env
== token: empty SUPERVISOR_API_TOKEN falls back
args: dns info
env: SUPERVISOR_API_TOKEN=
== config file via flag (yaml)
args: dns info --config {tmp}/cfg.yaml
unset: SUPERVISOR_ENDPOINT
unset: SUPERVISOR_TOKEN
response: data:{"a":1}
file: cfg.yaml = endpoint: {endpoint}\napi-token: file-token\nraw-json: true\n
== config file via flag (yml, quoted values, comments)
args: dns info --config {tmp}/cfg.yml
unset: SUPERVISOR_ENDPOINT
unset: SUPERVISOR_TOKEN
file: cfg.yml = # my config\nendpoint: "{endpoint}"   # dev\napi-token: 'file token'\n
== config file via flag (json)
args: dns info --config {tmp}/cfg.json
unset: SUPERVISOR_ENDPOINT
unset: SUPERVISOR_TOKEN
file: cfg.json = {"endpoint": "{endpoint}", "api-token": "json-token", "raw-json": true}
== config file: env beats file
args: dns info --config {tmp}/cfg.yaml
file: cfg.yaml = endpoint: 127.0.0.1:1\napi-token: file-token\n
== config file: flag beats file
args: dns info --config {tmp}/cfg.yaml --api-token flag-token
unset: SUPERVISOR_TOKEN
file: cfg.yaml = api-token: file-token\n
== config file missing
args: dns info --config {tmp}/nope.yaml
== config file in HOME (yaml)
args: dns info
env: HOME={tmp}
unset: SUPERVISOR_TOKEN
file: .homeassistant.yaml = api-token: home-token\n
== config file in HOME (yml)
args: dns info
env: HOME={tmp}
unset: SUPERVISOR_TOKEN
file: .homeassistant.yml = api-token: home-token-yml\n
== config file in HOME (json)
args: dns info
env: HOME={tmp}
unset: SUPERVISOR_TOKEN
file: .homeassistant.json = {"api-token": "home-json"}
== SUPERVISOR_CONFIG env is not used to locate the file
args: dns info
env: SUPERVISOR_CONFIG={tmp}/cfg.yaml
unset: SUPERVISOR_TOKEN
file: cfg.yaml = api-token: file-token\n
== config file uppercase keys and bool as number
args: dns info --config {tmp}/cfg.yaml
unset: SUPERVISOR_TOKEN
response: data:{"a":1}
file: cfg.yaml = API-TOKEN: upper\nRaw-Json: 1\n
== log level debug (stdout/requests only)
args: dns info --log-level debug
compare: stdout exit requests
== log level garbage
args: dns info --log-level nonsense
== log level env error
args: dns info
env: SUPERVISOR_LOG_LEVEL=error
